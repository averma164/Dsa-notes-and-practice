# 2095. Delete the Middle Node of a Linked List

You are given the head of a linked list. Delete the middle node, and return the head of the modified linked list.

# Approach :-

1) Cheack if head is NULL or head->next is NULL if yes :
    - return NULL
2) Initialize ListNode pointers :
    - slow = head
    - fast = head
    - prev = NULL
3) Start a while loop with condition fast != NULL && fast->next != NULL if yes update:
    - prev = slow
    - slow = slow->next
    - fast = fast->next->next
4) when the lopp ends connect:
    - prev->next = slow->next
    - delete slow
5) return head

# Complexity :-

- Time Complexity :- O(n)
- Space Complexity :- O(1)

# Program :-

class Solution {
public:
    ListNode* deleteMiddle(ListNode* head) {
        if (head == NULL || head->next == NULL) {
            return NULL;
        }
        ListNode* slow = head;
        ListNode* fast = head;
        ListNode* prev = NULL;
        while (fast != NULL && fast->next != NULL) {
            prev = slow;
            slow = slow->next;
            fast = fast->next->next;
        }
        prev->next = slow->next;
        delete slow;
        return head;
    }
};
